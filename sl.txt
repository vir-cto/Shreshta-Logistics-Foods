Zero Firebase Storage costs for millions of invoices.

Invoices are always 100% up-to-date with the database.

2. Cloud Firestore Security Rules (firestore.rules)
Paste these production security rules into your Firebase Console under Firestore Database -> Rules:

JavaScript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions for Role-Based Access Control
    function isAuthenticated() {
      return request.auth != null;
    }
    
    function getUserData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function isSuperAdmin() {
      return isAuthenticated() && getUserData().role == 'SUPER_ADMIN';
    }

    function isCoLoader() {
      return isAuthenticated() && (getUserData().role == 'CO_LOADER' || getUserData().role == 'SUPER_ADMIN');
    }

    // Users Collection
    match /users/{userId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin();
    }

    // Logistics AWBs Collection
    match /logistics_awbs/{awbId} {
      allow read: if true; // Public tracking access
      allow create, update: if isCoLoader();
      allow delete: if isSuperAdmin();
    }

    // Fuel Surcharges Master Collection
    match /fuel_surcharges/{surchargeId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin(); // Super Admin only management
    }

    // Master Tracking Stages Collection
    match /tracking_stages_master/{stageId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin();
    }

    // Food Products & Categories
    match /food_products/{productId} {
      allow read: if true; // Public storefront browsing
      allow write: if isSuperAdmin();
    }
    
    match /food_categories/{catId} {
      allow read: if true;
      allow write: if isSuperAdmin();
    }

    // Food Orders Collection
    match /food_orders/{orderId} {
      allow read: if true; // Public order status tracking
      allow create: if true; // Public checkout creation
      allow update: if isCoLoader();
      allow delete: if isSuperAdmin();
    }

    // Sender & Receiver Master Address Books
    match /address_book_senders/{id} {
      allow read, write: if isCoLoader();
    }

    match /address_book_receivers/{id} {
      allow read, write: if isCoLoader();
    }
  }
}
